  function infoMoeda(cod) {
    for (var i = 0; i < RESPOSTA.moedas.length; i++) {
      if (RESPOSTA.moedas[i].codigo === cod) return RESPOSTA.moedas[i];
    }
    return { codigo: cod, simbolo: cod.toUpperCase(), nome: cod.toUpperCase(), primeira: '' };
  }

  function dinheiro(centavos, cod) {
    var m = infoMoeda(cod || moeda);
    return m.simbolo + ' ' + (centavos / 100).toLocaleString('pt-BR',
      { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  }

  /* "2026-01" vira "janeiro de 2026". A chave do endpoint é ordenável como
     texto de propósito, e a tradução para gente acontece só aqui. */
  function mesPorExtenso(chave) {
    var p = String(chave).split('-');
    return MES_LONGO[(+p[1] || 1) - 1] + ' de ' + p[0];
  }

  /* Um mês do período conta? A regra é a mesma para todos os blocos, então
     ela mora num lugar só. */
  function noPeriodo(chave) {
    var ano = +String(chave).slice(0, 4);
    var mes = +String(chave).slice(5, 7) - 1;
    if (mesAberto !== null) return ano === anoAberto && mes === mesAberto;
    if (atalho === 'tudo') return true;
    return ano === anoAberto;
  }

  function somaDaMoeda(cod) {
    var s = { assinatura: 0, recarga: 0, outro: 0, taxa: 0, reembolso: 0,
              cobrancas: 0, meses: 0 };
    var m = RESPOSTA.meses[cod] || {};
    Object.keys(m).forEach(function (k) {
      if (!noPeriodo(k)) return;
      var v = m[k];
      s.assinatura += v.assinatura; s.recarga += v.recarga; s.outro += v.outro;
      s.taxa += v.taxa; s.reembolso += v.reembolso; s.cobrancas += v.cobrancas;
      s.meses++;
    });
    s.bruto = s.assinatura + s.recarga + s.outro;
    s.liquido = s.bruto - s.taxa;
    return s;
  }

  function nomeDoPeriodo() {
    if (mesAberto !== null) return MES_LONGO[mesAberto] + ' de ' + anoAberto;
    if (atalho === 'tudo') return 'o começo';
    if (atalho === 'ano') return anoAberto;
    return 'nos últimos 12 meses';
  }

  }

  function desenharMeses() {
    var caixa = d.querySelector('[data-meses]');

    /* COM TODAS AS MOEDAS, A RÉGUA MEDE COBRANÇAS, E NÃO VALOR.
       Altura de barra pede uma escala, e real e dólar não têm uma em comum.
       Cobrança tem: ela é contável e vale o mesmo em qualquer moeda. Então a
       régua troca de assunto e passa a mostrar o ritmo do mês, com o rótulo
       dizendo isso ao lado. Empilhar valores de moedas diferentes na mesma
       barra daria um desenho bonito e sem significado nenhum. */

    var maior = 0;
    MES_NOME.forEach(function (_, i) {
      var chave = anoAberto + '-' + String(i + 1).padStart(2, '0');
      var v = 0, existe = false;
      (todas ? RESPOSTA.moedas.map(function (m) { return m.codigo; }) : [moeda])
        .forEach(function (cod) {
          var linha = (RESPOSTA.meses[cod] || {})[chave];
          if (!linha) return;
          existe = true;
          v += todas ? linha.cobrancas
            : linha.assinatura + linha.recarga + linha.outro;
        });
      alturas[i] = existe ? v : null;
      if (v > maior) maior = v;
    });

    caixa.innerHTML = '';
    MES_NOME.forEach(function (nome, i) {
      var v = alturas[i];
      var b = d.createElement('button');
      b.className = 'per__mes' + (mesAberto === i ? ' ativo' : '');
      b.type = 'button';
      // Mês sem cobrança não aceita clique: abrir um mês vazio só troca a
      // tela toda por zeros, e isso não é uma resposta que alguém pediu.
      if (v === null) b.disabled = true;
      var alt = v && maior ? Math.round((v / maior) * 100) : 0;
      b.innerHTML = '<i style="--h:' + (alt ? alt + '%' : '2px') + '"></i><b>' + nome + '</b>';
      if (v !== null) {
        b.title = MES_LONGO[i] + ': ' + (todas
          ? v + (v === 1 ? ' cobrança' : ' cobranças')
          : dinheiro(v));

             de ida e volta, sem precisar procurar um "limpar". */
          mesAberto = (mesAberto === i) ? null : i;
          if (mesAberto !== null) atalho = 'ano';
          pintar();

  function pintar() {
    var todas = moeda === TODAS;
    desenharMoedas();
    desenharMeses();

    d.querySelector('[data-ano-todo]').textContent = anoAberto;
    /* Os limites de ano saem dos dados: não dá para navegar até um ano que o
       sistema nunca viu, e o ano corrente é a fronteira do outro lado. */
    var primeiro = HOJE_ANO;
    RESPOSTA.moedas.forEach(function (m) {
      var a = +String(m.primeira).slice(0, 4);
      if (a && a < primeiro) primeiro = a;
    });
    d.querySelector('[data-ano-passo="-1"]').disabled = anoAberto <= primeiro;
    d.querySelector('[data-ano-passo="1"]').disabled = anoAberto >= HOJE_ANO;
    d.querySelectorAll('[data-atalho]').forEach(function (b) {
      b.classList.toggle('ativo', mesAberto === null && b.dataset.atalho === atalho);
    });
    /* O atalho diz a data em vez de "o começo": "desde o começo" faz quem lê
       parar para calcular de quando é esse começo, e a resposta cabe no
       próprio botão. */
    d.querySelector('[data-atalho="tudo"]').textContent = 'Desde ' + desdeQuando();

    // ── com todas as moedas, a tabela toma o lugar do número grande ──
    d.querySelector('.din-topo').classList.toggle('din-topo--todas', todas);
    d.querySelector('[data-din-so-uma]').hidden = todas;
    d.querySelector('[data-din-todas]').hidden = !todas;
    d.querySelector('[data-card-fonte]').hidden = todas;
    if (todas) {
      desenharTodas();
      d.querySelector('[data-todas-quando]').textContent =
        mesAberto !== null || atalho === 'ano' ? 'Em ' + nomeDoPeriodo() + '.'
          : atalho === 'tudo' ? 'Desde ' + desdeQuando() + '.'
          : 'Nos últimos 12 meses.';
      return;
    }

    var s = somaDaMoeda(moeda);
    var quando = nomeDoPeriodo();

    d.querySelector('[data-din-rotulo]').textContent =
      mesAberto !== null || atalho === 'ano' ? 'Entrou em ' + quando
        : atalho === 'tudo' ? 'Entrou desde ' + desdeQuando() : 'Entrou nos últimos 12 meses';
    d.querySelector('[data-din-num]').innerHTML =
      dinheiro(verLiquido ? s.liquido : s.bruto);

    /* A comparação só faz sentido com um mês aberto: comparar "o ano todo"
       com o quê? Sem mês, a linha vira a média mensal do período, que é a
       leitura que serve para os dois. */
    var caixaVar = d.querySelector('[data-din-var]');
    var chaveAnt = mesAberto !== null && mesAberto > 0
      ? anoAberto + '-' + String(mesAberto).padStart(2, '0') : null;
    var ant = chaveAnt ? (RESPOSTA.meses[moeda] || {})[chaveAnt] : null;
    if (mesAberto !== null && ant) {
      var antes = ant.assinatura + ant.recarga + ant.outro - (verLiquido ? ant.taxa : 0);
      var difer = (verLiquido ? s.liquido : s.bruto) - antes;
      caixaVar.hidden = false;
      caixaVar.classList.toggle('din-var--sobe', difer >= 0);
      caixaVar.classList.toggle('din-var--desce', difer < 0);
      caixaVar.querySelector('svg path').setAttribute('d',
        difer >= 0 ? 'M3 11l5-5 5 5' : 'M3 5l5 5 5-5');
      caixaVar.querySelector('[data-din-var-valor]').innerHTML =
        (difer >= 0 ? '' : '− ') + dinheiro(Math.abs(difer));
      caixaVar.querySelector('[data-din-var-txt]').textContent =
        'em relação a ' + MES_LONGO[mesAberto - 1];
    } else if (mesAberto !== null) {
      caixaVar.hidden = true;
    } else {
      caixaVar.hidden = false;
      caixaVar.classList.add('din-var--sobe');
      caixaVar.classList.remove('din-var--desce');
      caixaVar.querySelector('svg path').setAttribute('d', 'M3 11l5-5 5 5');
      /* A média segue a aba: número líquido em cima e média bruta embaixo
         seriam dois valores diferentes do mesmo período, lado a lado. */
      var total = verLiquido ? s.liquido : s.bruto;
      caixaVar.querySelector('[data-din-var-valor]').innerHTML =
        dinheiro(s.meses ? Math.round(total / s.meses) : 0);
      caixaVar.querySelector('[data-din-var-txt]').textContent = 'por mês, na média';
    }

    var taxa = d.querySelector('[data-din-taxa]');
    taxa.hidden = !verLiquido;
    if (verLiquido) {
      d.querySelector('[data-taxa-bruto]').innerHTML = dinheiro(s.bruto);
      d.querySelector('[data-taxa-valor]').innerHTML = '− ' + dinheiro(s.taxa);
      d.querySelector('[data-taxa-liquido]').innerHTML = dinheiro(s.liquido);
      d.querySelector('[data-taxa-pct]').textContent =
        s.bruto ? (s.taxa / s.bruto * 100).toFixed(2).replace('.', ',') + '%' : '';
    }

    d.querySelector('[data-din-nota]').textContent = verLiquido
      ? 'O que o Stripe reteve em cada cobrança, somado. A taxa muda por cobrança: cartão de fora do Brasil e parcelamento custam mais que o cartão nacional à vista. Imposto não entra nesta conta.'
      : 'Tudo que foi pago no período, assinatura mais recarga. É dinheiro que entrou de verdade, e não cobrança em aberto.';

    // ── de onde vem, no mesmo período e na mesma moeda ──
    var pa = s.bruto ? (s.assinatura / s.bruto * 100) : 0;
    d.querySelector('[data-fonte-barra-a]').style.width = pa + '%';
    d.querySelector('[data-fonte-barra-b]').style.width = (100 - pa) + '%';
    d.querySelector('[data-fonte-assinatura]').innerHTML = dinheiro(s.assinatura);
    d.querySelector('[data-fonte-recarga]').innerHTML = dinheiro(s.recarga);
    var linhaOutro = d.querySelector('[data-fonte-outro-linha]');
    /* "Outro" é cobrança que entrou e não bate com assinatura nem com recarga
       nossa. A linha só aparece quando existe: numa tela onde ela é zero,
       mostrá-la seria fazer a pessoa procurar um problema que não tem. */
    linhaOutro.hidden = !s.outro;
    if (s.outro) d.querySelector('[data-fonte-outro]').innerHTML = dinheiro(s.outro);
    d.querySelector('[data-fonte-quando]').textContent =
      mesAberto !== null || atalho === 'ano' ? 'Em ' + quando + ', em ' + infoMoeda(moeda).nome.toLowerCase() + '.'
        : atalho === 'tudo' ? 'Desde ' + desdeQuando() + ', em ' + infoMoeda(moeda).nome.toLowerCase() + '.'
        : 'Nos últimos 12 meses, em ' + infoMoeda(moeda).nome.toLowerCase() + '.';
  }

  d.querySelectorAll('[data-ano-passo]').forEach(function (b) {
    b.addEventListener('click', function () {
      anoAberto += +b.dataset.anoPasso;
      mesAberto = null;
      if (atalho === 'tudo') atalho = 'ano';
      pintar();
    });
  });
  d.querySelector('[data-ano-todo]').addEventListener('click', function () {
    mesAberto = null; atalho = 'ano'; pintar();
  });
  d.querySelectorAll('[data-atalho]').forEach(function (b) {
    b.addEventListener('click', function () {
